"""
BUTTER PROTOCOL v14.0 – THE ORDER BOOK HUNTER
- Triangular arbitrage over 30 cycles (BTC, ETH, SOL, LINK, ADA, MATIC).
- Simulates via L2 order books to get REAL execution prices.
- Ranks cycles by estimated net profit.
- Cross-Exchange scanner (Kraken ↔ Coinbase) kept as secondary.
- Persistent ledger + nuclear keep-alive.

import base64
import hashlib
import hmac
import json
import os

import time

from urllib.parse import urlencode
from itertools import permutations

import requests

# ---------- PERSISTENT LEDGER ----------
LEDGER_FILE = "ledger.json"

def load_ledger():
    try:
        with open(LEDGER_FILE, "r") as f:
            data = json.load(f)
            return Decimal(data.get("profit", "0.00")), int(data.get("trades", 0))
    except:
        return Decimal("0.00"), 0

def save_ledger(profit, trades):
    with open(LEDGER_FILE, "w") as f:
        json.dump({"profit": str(profit), "trades": trades}, f)

# ---------- Kraken API Wrappers ----------
KRAKEN_BASE_URL = "https://api.kraken.com"
REQUEST_TIMEOUT = 10
session = requests.Session()

def D(v): return Decimal(str(v))

def kraken_private_request(api_key, api_secret, path, data):
    nonce = str(time.time_ns())
    payload = {"nonce": nonce, **data}
    postdata = urlencode(payload)
    encoded = (nonce + postdata).encode()
    message = path.encode() + hashlib.sha256(encoded).digest()
    decoded_secret = base64.b64decode(api_secret)
    api_sign = base64.b64encode(hmac.new(decoded_secret, message, hashlib.sha512).digest()).decode()
    resp = session.post(KRAKEN_BASE_URL + path, data=payload, headers={"API-Key": api_key, "API-Sign": api_sign}, timeout=REQUEST_TIMEOUT)
    resp.raise_for_status()
    result = resp.json()
    if result.get("error"):
        raise RuntimeError("Kraken: " + "; ".join(result["error"]))
    return result

def get_kraken_balance(api_key, api_secret):
    res = kraken_private_request(api_key, api_secret, "/0/private/Balance", {})
    return {k: D(v) for k, v in res["result"].items() if D(v) > 0}

def kraken_market_order(api_key, api_secret, pair, side, volume):
    return kraken_private_request(api_key, api_secret, "/0/private/AddOrder", {
        "pair": pair,
        "type": side.lower(),
        "ordertype": "market",
        "volume": format(volume, "f"),
    })

# ---------- ORDER BOOK FETCH & SIMULATION ----------
# Asset definitions
ASSETS = ["BTC", "ETH", "SOL", "LINK", "ADA", "MATIC"]
# Kraken pair naming (XBT = BTC)
USD_PAIRS = {
    "BTC": "XXBTZUSD",
    "ETH": "ETHUSD",
    "SOL": "SOLUSD",
    "LINK": "LINKUSD",
    "ADA": "ADAUSD",
    "MATIC": "MATICUSD",
}
CROSS_PAIRS = {
    ("BTC", "ETH"): "ETHXBT",
    ("BTC", "SOL"): "SOLXBT",
    ("BTC", "LINK"): "LINKXBT",
    ("BTC", "ADA"): "ADAXBT",
    ("BTC", "MATIC"): "MATICXBT",
    ("ETH", "SOL"): "SOLETH",  # Kraken has SOL/ETH
    ("ETH", "LINK"): "LINKETH",
    ("ETH", "ADA"): "ADAETH",
    ("ETH", "MATIC"): "MATICETH",
    ("SOL", "LINK"): "LINKSOL",
    ("SOL", "ADA"): "ADASOL",
    ("SOL", "MATIC"): "MATICSOL",
    ("LINK", "ADA"): "ADALINK",
    ("LINK", "MATIC"): "MATICLINK",
    ("ADA", "MATIC"): "MATICADA",
}
# Reverse mapping for lookup
CROSS_LOOKUP = {}
for (a, b), pair in CROSS_PAIRS.items():
    CROSS_LOOKUP[(a, b)] = pair
    CROSS_LOOKUP[(b, a)] = pair  # same pair, just reverse side

def fetch_order_book(pair, count=10):
    """Fetches L2 order book for a pair."""
    try:
        r = requests.get(f"https://api.kraken.com/0/public/Depth?pair={pair}&count={count}", timeout=3)
        data = r.json()
        if data.get("error"):
            return None
        res = list(data["result"].values())[0]
        asks = [[D(price), D(vol)] for price, vol in res["asks"]]
        bids = [[D(price), D(vol)] for price, vol in res["bids"]]
        return {"asks": asks, "bids": bids}
    except:
        return None

def simulate_triangular_cycle(asset_a, asset_b, start_usd, orderbooks, fees_pct):
    """
    Simulates: USD -> A -> B -> USD using actual order books.
    Returns: (profit_usd, avg_price_1, avg_price_2, avg_price_3, max_volume)
    """
    # Leg 1: Buy A with USD (using USD/A asks)
    pair_a_usd = USD_PAIRS.get(asset_a)
    if not pair_a_usd or pair_a_usd not in orderbooks:
        return None
    ob_a_usd = orderbooks[pair_a_usd]
    if not ob_a_usd:
        return None
    remaining_usd = start_usd
    a_amt = D("0")
    avg_price_1 = D("0")
    for price, vol in ob_a_usd["asks"]:
        cost = price * vol
        if remaining_usd <= cost:
            a_amt += remaining_usd / price
            avg_price_1 = (avg_price_1 + (remaining_usd / price) * price) / (a_amt) if a_amt > 0 else price
            remaining_usd = D("0")
            break
        else:
            a_amt += vol
            avg_price_1 = (avg_price_1 + vol * price) / a_amt
            remaining_usd -= cost
    if a_amt == 0:
        return None

    # Leg 2: Sell A for B (using A/B asks)
    pair_a_b = CROSS_LOOKUP.get((asset_a, asset_b))
    if not pair_a_b or pair_a_b not in orderbooks:
        return None
    ob_a_b = orderbooks[pair_a_b]
    if not ob_a_b:
        return None
    b_amt = D("0")
    avg_price_2 = D("0")
    # Determine if pair is quoted in B or A. Kraken pairs like SOLXBT are quoted in BTC (base is SOL).
    # We want to buy B with A. If pair is A/B, ask shows how much B for 1 A.
    # Actually, Kraken pairs are like SOLXBT: base = SOL, quote = XBT. Price = XBT per SOL.
    # So if we want to sell A for B, we are selling A and receiving B.
    # The 'ask' price is how much B you pay for 1 A. We want to sell A, so we fill the 'bid' side?
    # Wait: To sell A (base) and receive B (quote), we hit the BID side of A/B.
    # Bid price = how much quote (B) you get for 1 base (A).
    # So we use the BID side of the order book for A/B.
    bids_ab = ob_a_b["bids"]
    if not bids_ab:
        return None
    remaining_a = a_amt
    for price, vol in bids_ab:
        # price is B per 1 A. vol is in A.
        if remaining_a <= vol:
            b_amt += remaining_a * price
            avg_price_2 = (avg_price_2 + (remaining_a * price)) / (b_amt) if b_amt > 0 else price
            remaining_a = D("0")
            break
        else:
            b_amt += vol * price
            avg_price_2 = (avg_price_2 + vol * price) / b_amt
            remaining_a -= vol
    if b_amt == 0:
        return None

    # Leg 3: Sell B for USD (using B/USD bids)
    pair_b_usd = USD_PAIRS.get(asset_b)
    if not pair_b_usd or pair_b_usd not in orderbooks:
        return None
    ob_b_usd = orderbooks[pair_b_usd]
    if not ob_b_usd:
        return None
    usd_final = D("0")
    avg_price_3 = D("0")
    remaining_b = b_amt
    for price, vol in ob_b_usd["bids"]:
        # price is USD per 1 B. vol is in B.
        if remaining_b <= vol:
            usd_final += remaining_b * price
            avg_price_3 = (avg_price_3 + (remaining_b * price)) / (usd_final) if usd_final > 0 else price
            remaining_b = D("0")
            break
        else:
            usd_final += vol * price
            avg_price_3 = (avg_price_3 + vol * price) / usd_final
            remaining_b -= vol

    if usd_final == 0:
        return None

    profit = usd_final - start_usd
    # Apply fees (rough estimate – each leg ~0.16% maker, but we use taker for simulation safety)
    profit = profit * (D("1") - fees_pct / D("100"))
    return profit, avg_price_1, avg_price_2, avg_price_3, a_amt

# ---------- Streamlit UI ----------
st.set_page_config(page_title="🐱 ORDER BOOK HUNTER", layout="wide")
st.title("🐱 v14.0 – Order Book Hunter (Kraken)")

loaded_profit, loaded_trades = load_ledger()

if "running" not in st.session_state: st.session_state.running = False
if "live_mode" not in st.session_state: st.session_state.live_mode = False
if "logs" not in st.session_state: st.session_state.logs = []
if "trade_count" not in st.session_state: st.session_state.trade_count = loaded_trades
if "total_profit_usd" not in st.session_state: st.session_state.total_profit_usd = loaded_profit
if "last_trade_time" not in st.session_state: st.session_state.last_trade_time = 0.0

def add_log(msg):
    ts = time.strftime("%H:%M:%S")
    st.session_state.logs.insert(0, f"[{ts}] {msg}")
    st.session_state.logs = st.session_state.logs[:200]

# Sidebar
st.sidebar.header("🔐 Kraken API")
kraken_key = st.sidebar.text_input("API Key", type="password", key="kraken_key")
kraken_secret = st.sidebar.text_area("Private Key", type="password", key="kraken_secret", height=80)

st.sidebar.divider()
st.sidebar.header("⚙️ Strategy Settings")
min_profit_usd = D(st.sidebar.number_input("Min Profit ($)", value=0.01, step=0.01, format="%.2f"))
trade_size_usd = D(st.sidebar.number_input("Trade Size ($)", value=10.0, step=5.0, format="%.1f"))
tri_fees = D(st.sidebar.number_input("Estimated Fees (%)", value=0.50, step=0.01, format="%.2f"))
cooldown = st.sidebar.number_input("Cooldown (sec)", value=30, step=5)
poll = st.sidebar.slider("Refresh (sec)", 2, 15, 5)

st.sidebar.divider()
st.sidebar.header("🔴 Live Trading")
st.session_state.live_mode = st.sidebar.toggle("Enable live orders", value=st.session_state.live_mode)
if st.session_state.live_mode:
    st.sidebar.error("⚠️ LIVE ACTIVE")
else:
    st.sidebar.info("🧪 DRY RUN")

if st.sidebar.button("⏹ STOP BOT", use_container_width=True):
    st.session_state.running = False
    save_ledger(st.session_state.total_profit_usd, st.session_state.trade_count)

creds_ok = all([kraken_key.strip(), kraken_secret.strip()])

# Self-ping keepalive
def self_ping():
    slug = os.environ.get("REPL_SLUG", "")
    owner = os.environ.get("REPL_OWNER", "")
    if slug and owner:
        url = f"https://{slug}.{owner}.replit.app"
    else:
        url = "http://localhost:8080"
    while True:
        time.sleep(120)
        try:
            requests.get(url, timeout=5)
        except:
            pass

if creds_ok and st.session_state.running:
    threading.Thread(target=self_ping, daemon=True).start()

# Main controls
col1, col2 = st.columns(2)
with col1:
    if st.button("▶ START", use_container_width=True, disabled=not creds_ok or st.session_state.running):
        st.session_state.running = True
        st.rerun()
with col2:
    if st.button("Test Kraken", use_container_width=True, disabled=not creds_ok):
        try:
            bal = get_kraken_balance(kraken_key, kraken_secret)
            st.success("✅ Connected.")
            st.json({k: f"{v:.4f}" for k, v in bal.items() if v > 0})
            add_log("Kraken test passed.")
        except Exception as e:
            st.error(f"❌ {e}")
            add_log(f"Test failed: {e}")

# ---------- MAIN LOOP ----------
placeholder = st.empty()

if st.session_state.running and creds_ok:
    try:
        while st.session_state.running:
            # 1. Fetch ALL necessary order books
            pairs_to_fetch = list(USD_PAIRS.values()) + list(CROSS_PAIRS.values())
            orderbooks = {}
            for pair in pairs_to_fetch:
                ob = fetch_order_book(pair, count=8)  # limit depth to save bandwidth
                if ob:
                    orderbooks[pair] = ob

            # 2. Generate all triangular cycles (A -> B where A != B)
            cycles = []
            for asset_a, asset_b in permutations(ASSETS, 2):
                # Check if we have all pairs
                pair_a_usd = USD_PAIRS.get(asset_a)
                pair_b_usd = USD_PAIRS.get(asset_b)
                pair_a_b = CROSS_LOOKUP.get((asset_a, asset_b))
                if not pair_a_usd or not pair_b_usd or not pair_a_b:
                    continue
                if pair_a_usd not in orderbooks or pair_b_usd not in orderbooks or pair_a_b not in orderbooks:
                    continue
                # Simulate
                result = simulate_triangular_cycle(asset_a, asset_b, trade_size_usd, orderbooks, tri_fees)
                if result:
                    profit, avg1, avg2, avg3, vol = result
                    if profit > min_profit_usd:
                        cycles.append({
                            "route": f"USD→{asset_a}→{asset_b}→USD",
                            "asset_a": asset_a,
                            "asset_b": asset_b,
                            "profit": profit,
                            "avg1": avg1,
                            "avg2": avg2,
                            "avg3": avg3,
                            "vol": vol,
                        })

            # 3. Sort by profit descending
            cycles.sort(key=lambda x: x["profit"], reverse=True)

            # 4. Display UI
            with placeholder.container():
                st.subheader("📊 LIVE TRIANGULAR OPPORTUNITY BOARD (Order Book Simulated)")

                if cycles:
                    # Show top 10
                    df_data = []
                    for c in cycles[:10]:
                        df_data.append({
                            "Route": c["route"],
                            "Est. Profit ($)": f"${c['profit']:.4f}",
                            "Avg Buy": f"{c['avg1']:.2f} {c['asset_a']}",
                            "Avg Cross": f"{c['avg2']:.6f}",
                            "Avg Sell": f"{c['avg3']:.2f} USD",
                            "Vol": f"{c['vol']:.6f}",
                        })
                    st.table(df_data)

                    # 5. Auto-execute best cycle if live
                    best = cycles[0]
                    if best["profit"] > min_profit_usd and (time.time() - st.session_state.last_trade_time) >= cooldown:
                        if st.session_state.live_mode:
                            st.success(f"🔥 EXECUTING TOP CYCLE: {best['route']} | Profit: ${best['profit']:.4f}")
                            try:
                                # Execute market orders (we simulated the exact volume)
                                # Leg 1: Buy A with USD
                                kraken_market_order(kraken_key, kraken_secret, USD_PAIRS[best["asset_a"]], "buy", best["vol"])
                                # Leg 2: Sell A for B
                                pair_ab = CROSS_LOOKUP[(best["asset_a"], best["asset_b"])]
                                kraken_market_order(kraken_key, kraken_secret, pair_ab, "sell", best["vol"])
                                # Leg 3: Sell B for USD
                                b_amt = best["vol"] * best["avg2"]  # approximate B amount
                                kraken_market_order(kraken_key, kraken_secret, USD_PAIRS[best["asset_b"]], "sell", b_amt)

                                st.session_state.total_profit_usd += best["profit"]
                                st.session_state.trade_count += 1
                                st.session_state.last_trade_time = time.time()
                                save_ledger(st.session_state.total_profit_usd, st.session_state.trade_count)
                                add_log(f"✅ TRI #{st.session_state.trade_count} | {best['route']} | profit ${best['profit']:.4f}")
                                st.balloons()
                            except Exception as e:
                                st.error(f"Execution error: {e}")
                                add_log(f"Execution error: {e}")
                        else:
                            st.warning(f"🔍 DRY RUN: Best cycle {best['route']} profit ${best['profit']:.4f}")
                            st.session_state.last_trade_time = time.time()
                else:
                    st.info("No profitable cycles found with current depth and fees.")

                st.subheader("📋 Activity Log")
                st.code("\n".join(st.session_state.logs[:30]) or "No activity.")

                st.subheader("📊 Cross-Exchange Scanner (Kraken ↔ Coinbase)")
                # Simple cross-exchange snapshot for SOL (keep it light)
                try:
                    r_k = requests.get("https://api.kraken.com/0/public/Ticker?pair=SOLUSD", timeout=2)
                    r_c = requests.get("https://api.coinbase.com/v2/prices/SOL-USD/spot", timeout=2)
                    k_price = D(list(r_k.json()["result"].values())[0]["c"][0])
                    c_price = D(r_c.json()["data"]["amount"])
                    diff = (abs(k_price - c_price) / min(k_price, c_price)) * 100
                    st.metric("SOL Spread (Kraken vs Coinbase)", f"{diff:.3f}%", delta=f"K: ${k_price:.2f} | C: ${c_price:.2f}")
                except:
                    st.warning("Cross-exchange SOL data unavailable.")

            time.sleep(poll)

    except Exception as e:
        with placeholder.container():
            st.error(f"🚨 Loop crashed: {e}")
        add_log(f"Loop crash: {e}")
        st.session_state.running = False
        save_ledger(st.session_state.total_profit_usd, st.session_state.trade_count)

else:
    if not creds_ok:
        placeholder.warning("Paste your Kraken API keys and test connection.")
    else:
        placeholder.info("Bot idle. Press START to begin monitoring.")

st.divider()
st.caption("🐱 v14.0 – Simulates actual order book depth. No paper profits.")
