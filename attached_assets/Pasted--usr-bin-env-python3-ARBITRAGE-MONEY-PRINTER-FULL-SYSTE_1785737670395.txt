# ------------------------- DATA STRUCTURES -------------------------

class OrderBook:
    """Local order book with nanosecond timestamp."""
    def __init__(self):
        self.bids: List[Tuple[Decimal, Decimal]] = []  # (price, size)
        self.asks: List[Tuple[Decimal, Decimal]] = []
        self.timestamp_ns: int = 0
        self.last_update: float = 0.0

    def update(self, bids: List[Tuple[Decimal, Decimal]],
               asks: List[Tuple[Decimal, Decimal]]):
        self.bids = bids
        self.asks = asks
        self.timestamp_ns = time.time_ns()
        self.last_update = time.time()

    def best_bid(self) -> Decimal:
        return self.bids[0][0] if self.bids else Decimal('0')

    def best_ask(self) -> Decimal:
        return self.asks[0][0] if self.asks else Decimal('0')

    def spread_percent(self) -> float:
        bid = self.best_bid()
        ask = self.best_ask()
        if bid == 0 or ask == 0:
            return float('inf')
        return float((ask - bid) / bid * 100)

    def mid_price(self) -> Decimal:
        return (self.best_bid() + self.best_ask()) / 2

# ------------------------- WEBSOCKET CLIENTS -------------------------

class ExchangeWebSocket:
    """Base WebSocket client for an exchange."""
    def __init__(self, exchange_name: str):
        self.exchange = exchange_name
        self.url = EXCHANGE_WS_URLS[exchange_name]
        self.books: Dict[str, OrderBook] = defaultdict(OrderBook)
        self.connected = False
        self.ws = None
        self.subscription_task = None

    async def connect(self):
        """Establish WebSocket connection and subscribe to all pairs."""
        try:
            self.ws = await websockets.connect(self.url, ping_interval=20, ping_timeout=10)
            self.connected = True
            logger.info(f"Connected to {self.exchange}")
            await self.subscribe()
            return self.ws
        except Exception as e:
            logger.error(f"Connection failed for {self.exchange}: {e}")
            self.connected = False
            return None

    async def subscribe(self):
        """Send subscription message (to be overridden)."""
        raise NotImplementedError

    async def listen(self):
        """Listen for incoming messages and update books."""
        if not self.ws:
            return
        try:
            async for message in self.ws:
                await self.handle_message(message)
        except websockets.ConnectionClosed:
            logger.warning(f"Connection closed for {self.exchange}, reconnecting...")
            self.connected = False
            await self.reconnect()

    async def reconnect(self):
        """Attempt to reconnect with backoff."""
        for attempt in range(10):
            logger.info(f"Reconnecting {self.exchange} attempt {attempt+1}")
            if await self.connect():
                await self.listen()
                return
            await asyncio.sleep(2 ** attempt)
        logger.error(f"Failed to reconnect {self.exchange} after 10 attempts")

    async def handle_message(self, message: str):
        """Parse message and update book (to be overridden)."""
        raise NotImplementedError

    def get_book(self, pair: str) -> Optional[OrderBook]:
        return self.books.get(pair)

# ---------- Kraken ----------
class KrakenWS(ExchangeWebSocket):
    def __init__(self):
        super().__init__("kraken")

    async def subscribe(self):
        for pair in PAIRS:
            symbol = PAIR_MAP["kraken"][pair]
            sub_msg = {
                "method": "subscribe",
                "params": {
                    "channel": "book",
                    "symbol": symbol,
                    "depth": 10
                }
            }
            await self.ws.send(json.dumps(sub_msg))
        logger.info(f"{self.exchange} subscribed to {len(PAIRS)} pairs")

    async def handle_message(self, message: str):
        try:
            data = json.loads(message)
            if "data" in data and len(data["data"]) > 0:
                for entry in data["data"]:
                    symbol = entry.get("symbol")
                    if not symbol:
                        continue
                    # Find pair from symbol
                    pair = None
                    for p in PAIRS:
                        if PAIR_MAP["kraken"][p] == symbol:
                            pair = p
                            break
                    if not pair:
                        continue
                    book = self.books[pair]
                    bids = [(Decimal(b[0]), Decimal(b[1])) for b in entry.get("bids", [])]
                    asks = [(Decimal(a[0]), Decimal(a[1])) for a in entry.get("asks", [])]
                    book.update(bids, asks)
        except Exception as e:
            logger.debug(f"Kraken parse error: {e}")

# ---------- Coinbase ----------
class CoinbaseWS(ExchangeWebSocket):
    def __init__(self):
        super().__init__("coinbase")

    async def subscribe(self):
        # Coinbase uses channel "level2"
        sub_msg = {
            "type": "subscribe",
            "product_ids": [PAIR_MAP["coinbase"][p] for p in PAIRS],
            "channels": ["level2"]
        }
        await self.ws.send(json.dumps(sub_msg))
        logger.info(f"{self.exchange} subscribed")

    async def handle_message(self, message: str):
        try:
            data = json.loads(message)
            if data.get("type") == "l2update":
                product_id = data.get("product_id")
                pair = None
                for p in PAIRS:
                    if PAIR_MAP["coinbase"][p] == product_id:
                        pair = p
                        break
                if not pair:
                    return
                changes = data.get("changes", [])
                book = self.books[pair]
                # Coinbase sends incremental updates; we need to maintain full book
                # For simplicity, we'll replace with full snapshot if available
                # But we don't have snapshot here; we can either accumulate or rely on initial snapshot.
                # For demo, we'll just store the updates but we need a snapshot.
                # Instead, we'll just store the best bid/ask from changes.
                # Better: use REST to get full snapshot initially, but we'll skip for brevity.
                # For production, you should maintain order book with incremental updates.
                # Here we'll just store the best bid/ask from changes.
                best_bid = Decimal('0')
                best_ask = Decimal('0')
                for change in changes:
                    side, price, size = change
                    price = Decimal(price)
                    size = Decimal(size)
                    if side == "buy":
                        if price > best_bid:
                            best_bid = price
                    elif side == "sell":
                        if best_ask == 0 or price < best_ask:
                            best_ask = price
                if best_bid > 0 and best_ask > 0:
                    book.update([(best_bid, Decimal('1'))], [(best_ask, Decimal('1'))])
        except Exception as e:
            logger.debug(f"Coinbase parse error: {e}")

# ---------- Binance ----------
class BinanceWS(ExchangeWebSocket):
    def __init__(self):
        super().__init__("binance")

    async def subscribe(self):
        streams = [f"{PAIR_MAP['binance'][p]}@bookTicker" for p in PAIRS]
        sub_msg = {
            "method": "SUBSCRIBE",
            "params": streams,
            "id": 1
        }
        await self.ws.send(json.dumps(sub_msg))
        logger.info(f"{self.exchange} subscribed")

    async def handle_message(self, message: str):
        try:
            data = json.loads(message)
            if "data" in data:
                data = data["data"]
            if "s" in data:  # symbol
                symbol = data["s"].upper()
                pair = None
                for p in PAIRS:
                    if PAIR_MAP["binance"][p].upper() == symbol:
                        pair = p
                        break
                if not pair:
                    return
                bid = Decimal(data.get("b", 0))
                ask = Decimal(data.get("a", 0))
                if bid > 0 and ask > 0:
                    book = self.books[pair]
                    book.update([(bid, Decimal('1'))], [(ask, Decimal('1'))])
        except Exception as e:
            logger.debug(f"Binance parse error: {e}")

class Strategy:
    """Base class for arbitrage strategies."""
    def __init__(self, name: str):
        self.name = name

    async def find_opportunity(self, exchanges: Dict[str, ExchangeWebSocket]) -> Optional[Dict]:
        """Return dict with 'pair', 'edge', 'side', 'size', 'details' if found."""
        raise NotImplementedError

# ---------- Cross-Exchange Arbitrage ----------
class CrossExchangeArb(Strategy):
    def __init__(self):
        super().__init__("cross_exchange")
        self.min_spread = MIN_SPREAD_PERCENT

    async def find_opportunity(self, exchanges: Dict[str, ExchangeWebSocket]):
        # For each pair, compare best bid across exchanges and best ask across exchanges
        for pair in PAIRS:
            best_bid = Decimal('0')
            best_ask = Decimal('0')
            bid_exchange = None
            ask_exchange = None
            for ex_name, ex_client in exchanges.items():
                book = ex_client.get_book(pair)
                if not book:
                    continue
                bid = book.best_bid()
                ask = book.best_ask()
                if bid > best_bid:
                    best_bid = bid
                    bid_exchange = ex_name
                if ask > 0 and (best_ask == 0 or ask < best_ask):
                    best_ask = ask
                    ask_exchange = ex_name
            if best_bid > 0 and best_ask > 0 and bid_exchange and ask_exchange and bid_exchange != ask_exchange:
                spread = (best_bid - best_ask) / best_ask * 100
                # Consider fees and slippage
                net_edge = spread - (TRADING_FEE * 2) - SLIPPAGE_BUFFER * 2
                if net_edge > self.min_spread:
                    return {
                        "strategy": self.name,
                        "pair": pair,
                        "edge": net_edge,
                        "side": "buy_at_ask_sell_at_bid",
                        "buy_exchange": ask_exchange,
                        "sell_exchange": bid_exchange,
                        "buy_price": best_ask,
                        "sell_price": best_bid,
                        "details": f"Buy {pair} on {ask_exchange} at {best_ask}, sell on {bid_exchange} at {best_bid}"
                    }
        return None

# ---------- Triangular Arbitrage ----------
class TriangularArb(Strategy):
    def __init__(self):
        super().__init__("triangular")
        self.min_edge = 0.1  # 0.1%

    async def find_opportunity(self, exchanges: Dict[str, ExchangeWebSocket]):
        # For each exchange, check triangular loops (e.g., USDT -> SOL -> ETH -> USDT)
        for ex_name, ex_client in exchanges.items():
            # Need prices for SOL/USDT, ETH/USDT, ETH/SOL
            # We'll map pair names: we need to know the base/quote for each.
            # For simplicity, assume we have SOL/USDT, ETH/USDT, and ETH/SOL.
            # We'll create a price dict for that exchange.
            prices = {}
            for pair in PAIRS:
                book = ex_client.get_book(pair)
                if book:
                    # Use mid price
                    prices[pair] = book.mid_price()
            # Check triangle: USDT -> SOL -> ETH -> USDT
            if "SOL/USD" in prices and "ETH/USD" in prices and "ETH/SOL" in prices:
                sol_usd = prices["SOL/USD"]
                eth_usd = prices["ETH/USD"]
                eth_sol = prices["ETH/SOL"]  # ETH denominated in SOL
                # Path: 1 USDT -> SOL: 1/sol_usd SOL; then SOL -> ETH: (1/sol_usd) / eth_sol ETH; then ETH -> USDT: * eth_usd
                final = (1 / sol_usd) / eth_sol * eth_usd
                profit = (final - 1) * 100
                # Subtract fees (3 trades)
                net_profit = profit - (TRADING_FEE * 3 * 100)  # convert to percent
                if net_profit > self.min_edge:
                    return {
                        "strategy": self.name,
                        "pair": "USDT-SOL-ETH-USDT",
                        "edge": net_profit,
                        "side": "triangular",
                        "exchange": ex_name,
                        "details": f"USDT->SOL->ETH->USDT on {ex_name} profit {net_profit:.3f}%"
                    }
        return None

# ---------- Cointegration Arbitrage ----------
class CointegrationArb(Strategy):
    """Pairs trading using Kalman filter for dynamic hedge ratio."""
    def __init__(self, asset1: str, asset2: str):
        super().__init__(f"coint_{asset1}_{asset2}")
        self.asset1 = asset1
        self.asset2 = asset2
        self.asset1_price = None
        self.asset2_price = None
        self.kf = KalmanFilter(
            transition_matrices=np.eye(2),
            observation_matrices=np.ones((1, 2)),
            initial_state_mean=[1.0, 0.0],
            n_dim_obs=1
        )
        self.state_means = []
        self.spreads = []
        self.std_dev = 1.0
        self.initialized = False

    async def find_opportunity(self, exchanges: Dict[str, ExchangeWebSocket]):
        # For each exchange, check if both assets are traded, get mid prices
        # We'll use the first exchange that has both.
        for ex_name, ex_client in exchanges.items():
            pair1 = f"{self.asset1}/USD"
            pair2 = f"{self.asset2}/USD"
            book1 = ex_client.get_book(pair1)
            book2 = ex_client.get_book(pair2)
            if not book1 or not book2:
                continue
            price1 = book1.mid_price()
            price2 = book2.mid_price()
            if price1 == 0 or price2 == 0:
                continue

            # Update Kalman filter
            obs = np.array([float(price2)])
            if not self.initialized:
                self.kf = self.kf.em([obs])  # Initialize with first observation
                self.initialized = True
                self.state_means = [self.kf.initial_state_mean]
            else:
                filtered_state_mean, _ = self.kf.filter_update(
                    self.state_means[-1], observation=obs
                )
                self.state_means.append(filtered_state_mean)

            # Compute spread: price1 - hedge_ratio * price2
            hedge_ratio = self.state_means[-1][0]
            spread = float(price1) - hedge_ratio * float(price2)
            self.spreads.append(spread)
            if len(self.spreads) > 100:
                self.spreads = self.spreads[-100:]
                self.std_dev = np.std(self.spreads)
            else:
                self.std_dev = 1.0  # wait for enough data

            # Check if spread exceeds 2 standard deviations
            if len(self.spreads) > 20:
                z = (spread - np.mean(self.spreads)) / (self.std_dev + 1e-9)
                if abs(z) > 2.0:
                    # Generate signal
                    side = "buy_asset1_sell_asset2" if z < 0 else "sell_asset1_buy_asset2"
                    edge = abs(z) * 0.1  # rough edge estimate
                    # Subtract fees (two trades)
                    net_edge = edge - (TRADING_FEE * 2 * 100)
                    if net_edge > 0.05:  # minimum 0.05% edge
                        return {
                            "strategy": self.name,
                            "pair": f"{self.asset1}/{self.asset2}",
                            "edge": net_edge,
                            "side": side,
                            "exchange": ex_name,
                            "details": f"Spread z={z:.2f}, hedge ratio={hedge_ratio:.4f}"
                        }
        return None

# ------------------------- KELLY SIZER -------------------------

class KellySizer:
    def __init__(self, win_rate=ESTIMATED_WIN_RATE, fraction=KELLY_FRACTION):
        self.win_rate = win_rate
        self.fraction = fraction

    def calculate(self, edge_percent: float, bankroll: float) -> float:
        """Return position size in USD."""
        if edge_percent <= 0:
            return 0
        b = edge_percent / 100.0
        p = self.win_rate
        q = 1 - p
        f = (b * p - q) / b if b > 0 else 0
        f = max(0, min(f, self.fraction))
        return bankroll * f

# ------------------------- AI RISK AUDITOR -------------------------

class RiskAuditor:
    """Rule-based auditor with optional LLM sentiment."""
    def __init__(self):
        self.sentiment_cache = {}

    def approve(self, opportunity: Dict, exchanges: Dict) -> bool:
        """Check basic risk rules and optionally sentiment."""
        if not opportunity:
            return False
        # 1. Edge must be positive
        if opportunity.get('edge', 0) <= 0:
            return False
        # 2. Check liquidity: for cross-exchange, ensure size is not too large relative to order book depth.
        # For simplicity, we'll just check that the edge is above min threshold.
        # 3. Check market volatility (skipping for now)
        # 4. Optional sentiment check using a free API (mock)
        # We'll just approve if edge > 0.1%
        return opportunity['edge'] > 0.1

# ------------------------- MAIN ARBITRAGE ENGINE -------------------------

class MoneyPrinter:
    def __init__(self, dry_run=True):
        self.dry_run = dry_run
        self.exchanges: Dict[str, ExchangeWebSocket] = {
            "kraken": KrakenWS(),
            "coinbase": CoinbaseWS(),
            # "binance": BinanceWS()  # Uncomment if you have Binance keys
        }
        self.strategies = [
            CrossExchangeArb(),
            TriangularArb(),
            CointegrationArb("SOL", "ETH"),
            CointegrationArb("BTC", "ETH"),
            CointegrationArb("AVAX", "DOT")
        ]
        self.kelly = KellySizer()
        self.auditor = RiskAuditor()
        self.bankroll = 1000.0  # Starting capital in USD
        self.total_trades = 0
        self.total_profit = 0.0
        self.opportunity_log = []

    async def connect_all(self):
        tasks = []
        for ex in self.exchanges.values():
            tasks.append(ex.connect())
        await asyncio.gather(*tasks)
        # Start listening tasks
        listen_tasks = []
        for ex in self.exchanges.values():
            if ex.connected:
                listen_tasks.append(asyncio.create_task(ex.listen()))
        return listen_tasks

    async def run_loop(self):
        # Start WebSocket connections
        listen_tasks = await self.connect_all()
        logger.info("All exchanges connected, starting arbitrage loop")

        # Main loop
        while True:
            # Scan for opportunities across strategies
            tasks = []
            for strategy in self.strategies:
                tasks.append(strategy.find_opportunity(self.exchanges))
            results = await asyncio.gather(*tasks)

            for opp in results:
                if opp is None:
                    continue
                # Audit
                if not self.auditor.approve(opp, self.exchanges):
                    continue

                # Size using Kelly
                size_usd = self.kelly.calculate(opp['edge'], self.bankroll)
                if size_usd <= 0:
                    continue
                # Cap by max position
                size_usd = min(size_usd, MAX_POSITION_SIZE)

                # Execute trade
                if self.dry_run:
                    logger.info(f"DRY RUN: {opp['details']} | Edge: {opp['edge']:.3f}% | Size: ${size_usd:.2f}")
                    # Simulate profit
                    profit = size_usd * (opp['edge'] / 100.0)
                    self.bankroll += profit
                    self.total_profit += profit
                    self.total_trades += 1
                    self.opportunity_log.append(opp)
                else:
                    # Real execution would go here (place orders)
                    logger.info(f"EXECUTING: {opp['details']} | Size: ${size_usd:.2f}")
                    # For now, we just dry-run even if dry_run False
                    # In real implementation, call exchange API to place orders
                    pass

            # Log stats periodically
            if self.total_trades % 10 == 0:
                logger.info(f"Trades: {self.total_trades} | Bankroll: ${self.bankroll:.2f} | Profit: ${self.total_profit:.2f}")

            # Sleep 1ms to prevent CPU hog
            await asyncio.sleep(0.001)

    async def run(self):
        try:
            await self.run_loop()
        except KeyboardInterrupt:
            logger.info("Shutting down...")
        finally:
            # Close connections
            for ex in self.exchanges.values():
                if ex.ws:
                    await ex.ws.close()
