def kraken_sig(endpoint, data):
    nonce = str(int(time.time()*1000)); post_data = urlencode(data) if data else ""
    encoded = (nonce+post_data).encode()
    msg = endpoint.encode()+hashlib.sha256(encoded).digest()
    sig = base64.b64encode(hmac.new(base64.b64decode(KRAKEN_SEC),msg,hashlib.sha512).digest()).decode()
    return {"API-Key":KRAKEN_KEY,"API-Sign":sig}

async def kraken_cvd(session):
    async with session.get(f"{KRAKEN}/0/public/Trades?pair=ETHUSD",timeout=5) as r:
        data = await r.json()
    trades = data.get("result",{}).get("XETHZUSD",[]); cutoff = time.time()-CVD_WINDOW_S
    cvd = 0.0
    for t in (trades if isinstance(trades,list) else list(trades.values())):
        if isinstance(t,list) and len(t)>=4:
            ts,vol,side = t[2],float(t[1]),t[3]
            if ts>=cutoff: cvd += vol*(1 if side=="b" else -1)
    return cvd
