# --- MANUAL EXECUTION BUTTON ---
if top_cycle and top_cycle["profit"] > min_profit_usd:
    col_exec, col_info = st.columns([1, 2])
    with col_exec:
        if st.button("🔴 EXECUTE TOP ROUTE NOW", use_container_width=True, type="primary"):
            if not st.session_state.live_mode:
                st.warning("Enable Live Trading in sidebar first!")
            else:
                # Re-fetch fresh order books for the top route
                fresh_books = {}
                pairs_needed = [USD_PAIRS[top_cycle["asset_a"]], USD_PAIRS[top_cycle["asset_b"]]]
                cross_needed = get_cross_pair(top_cycle["asset_a"], top_cycle["asset_b"])
                if cross_needed not in orderbooks:
                    cross_needed = get_cross_pair(top_cycle["asset_b"], top_cycle["asset_a"])
                pairs_needed.append(cross_needed)
                for p in pairs_needed:
                    ob = fetch_order_book(p, count=10)
                    if ob:
                        fresh_books[p] = ob
                if all(p in fresh_books for p in pairs_needed):
                    fresh_result = simulate_triangular_cycle(
                        top_cycle["asset_a"], top_cycle["asset_b"],
                        trade_size_usd, fresh_books, tri_fees
                    )
                    if fresh_result and fresh_result[0] > min_profit_usd:
                        st.success(f"🔥 Executing {top_cycle['route']} | Size: ${trade_size_usd:.0f}")
                        try:
                            # Place market orders
                            kraken_market_order(kraken_key, kraken_secret, USD_PAIRS[top_cycle["asset_a"]], "buy", top_cycle["vol"])
                            pair_ab = get_cross_pair(top_cycle["asset_a"], top_cycle["asset_b"])
                            if pair_ab not in fresh_books:
                                pair_ab = get_cross_pair(top_cycle["asset_b"], top_cycle["asset_a"])
                            kraken_market_order(kraken_key, kraken_secret, pair_ab, "sell", top_cycle["vol"])
                            b_amt = top_cycle["vol"] * (top_cycle["profit"] / trade_size_usd)
                            kraken_market_order(kraken_key, kraken_secret, USD_PAIRS[top_cycle["asset_b"]], "sell", b_amt)
                            st.session_state.total_profit_usd += fresh_result[0]
                            st.session_state.trade_count += 1
                            save_ledger(st.session_state.total_profit_usd, st.session_state.trade_count)
                            add_log(f"✅ MANUAL TRI #{st.session_state.trade_count} | profit ${fresh_result[0]:.4f}")
                            st.balloons()
                        except Exception as e:
                            st.error(f"Execution error: {e}")
                            add_log(f"Execution error: {e}")
                    else:
                        st.error("Pre-flight failed – edge disappeared.")
                else:
                    st.error("Could not fetch fresh order books.")
    with col_info:
        st.info(f"Current best: {top_cycle['route']} | Profit: ${top_cycle['profit']:.4f}")
