# ---------- PERSISTENT LEDGER ----------
LEDGER_FILE = "ledger.json"

def load_ledger():
    try:
        with open(LEDGER_FILE, "r") as f:
            data = json.load(f)
            return Decimal(data.get("profit", "0.00")), int(data.get("trades", 0))
    except:
        return Decimal("0.00"), 0

def save_ledger(profit, trades):
    with open(LEDGER_FILE, "w") as f:
        json.dump({"profit": str(profit), "trades": trades}, f)

# ---------- WebSocket price store ----------
latest_prices = {"kraken": None, "coinbase": None}
price_lock = threading.Lock()

def on_kraken_message(ws, message):
    global latest_prices
    try:
        data = json.loads(message)
        if isinstance(data, list) and len(data) > 1 and data[2] == "ticker":
            tick = data[1]
            if "c" in tick:
                with price_lock:
                    latest_prices["kraken"] = Decimal(tick["c"][0])
    except:
        pass

def start_kraken_ws():
    ws = websocket.WebSocketApp("wss://ws.kraken.com/v2", on_message=on_kraken_message)
    def on_open(ws):
        ws.send(json.dumps({"method": "subscribe", "params": {"channel": "ticker", "symbol": ["SOL/USD"]}}))
    ws.on_open = on_open
    ws.run_forever()

def on_coinbase_message(ws, message):
    global latest_prices
    try:
        data = json.loads(message)
        if data.get("type") == "ticker" and data.get("product_id") == "SOL-USD":
            with price_lock:
                latest_prices["coinbase"] = Decimal(data["price"])
    except:
        pass

def start_coinbase_ws():
    ws = websocket.WebSocketApp("wss://ws-feed.exchange.coinbase.com", on_message=on_coinbase_message)
    def on_open(ws):
        ws.send(json.dumps({"type": "subscribe", "product_ids": ["SOL-USD"], "channels": ["ticker"]}))
    ws.on_open = on_open
    ws.run_forever()

# ---------- REST helpers ----------
def D(v): return Decimal(str(v))

def kraken_rest():
    try:

# ---------- Kraken API (unchanged) ----------
KRAKEN_BASE_URL = "https://api.kraken.com"
REQUEST_TIMEOUT = 10
session = requests.Session()

def kraken_private_request(api_key, api_secret, path, data):
    nonce = str(time.time_ns())
    payload = {"nonce": nonce, **data}
    postdata = urlencode(payload)
    encoded = (nonce + postdata).encode()
    message = path.encode() + hashlib.sha256(encoded).digest()
    decoded_secret = base64.b64decode(api_secret)
    api_sign = base64.b64encode(hmac.new(decoded_secret, message, hashlib.sha512).digest()).decode()
    resp = session.post(KRAKEN_BASE_URL + path, data=payload, headers={"API-Key": api_key, "API-Sign": api_sign}, timeout=REQUEST_TIMEOUT)
    resp.raise_for_status()
    result = resp.json()
    if result.get("error"):
        raise RuntimeError("Kraken: " + "; ".join(result["error"]))
    return result

def get_kraken_balance(api_key, api_secret):
    res = kraken_private_request(api_key, api_secret, "/0/private/Balance", {})
    return {k: D(v) for k, v in res["result"].items() if D(v) > 0}

def kraken_limit_order(api_key, api_secret, pair, side, volume, price):
    return kraken_private_request(api_key, api_secret, "/0/private/AddOrder", {
        "pair": pair,
        "type": side.lower(),
        "ordertype": "limit",
        "post_only": "true",
        "volume": format(volume, "f"),
        "price": format(price, "f"),
    })

def kraken_market_order(api_key, api_secret, pair, side, volume):
    return kraken_private_request(api_key, api_secret, "/0/private/AddOrder", {
        "pair": pair,
        "type": side.lower(),
        "ordertype": "market",
        "volume": format(volume, "f"),
    })

# ---------- TRIANGULAR ARBITRAGE SCANNER ----------
def scan_triangular():
    """Fetches bid/ask for BTC/USD, SOL/USD, SOL/BTC and calculates two routes."""
    try:
        # Fetch tickers
        r = requests.get("https://api.kraken.com/0/public/Ticker?pair=XXBTZUSD,SOLUSD,SOLXBT", timeout=5)
        data = r.json()
        if data.get("error"):
            return None, None, None
        res = data["result"]

        # BTC/USD
        btc_usd = res["XXBTZUSD"]
        btc_bid = D(btc_usd["b"][0])
        btc_ask = D(btc_usd["a"][0])

        # SOL/USD
        sol_usd = res["SOLUSD"]
        sol_bid = D(sol_usd["b"][0])
        sol_ask = D(sol_usd["a"][0])

        # SOL/BTC (how much BTC for 1 SOL)
        sol_btc = res["SOLXBT"]
        sol_btc_bid = D(sol_btc["b"][0])  # bid: buy SOL for BTC
        sol_btc_ask = D(sol_btc["a"][0])  # ask: sell SOL for BTC

        # ---- ROUTE 1: USD -> BTC -> SOL -> USD ----
        # Start with 1 USD
        start = D("1")
        # Buy BTC with USD
        btc_amt = start / btc_ask
        # Buy SOL with BTC (pay in BTC)
        sol_amt = btc_amt / sol_btc_ask
        # Sell SOL for USD
        usd_final = sol_amt * sol_bid
        profit_1 = (usd_final - start) / start * D("100")

        # ---- ROUTE 2: USD -> SOL -> BTC -> USD ----
        # Buy SOL with USD
        sol_amt2 = start / sol_ask
        # Sell SOL for BTC
        btc_amt2 = sol_amt2 * sol_btc_bid
        # Sell BTC for USD
        usd_final2 = btc_amt2 * btc_bid
        profit_2 = (usd_final2 - start) / start * D("100")

        # Build route descriptions
        route1 = "USD → BTC → SOL → USD"
        route2 = "USD → SOL → BTC → USD"

        return {
            "route1": {"name": route1, "profit_pct": profit_1, "detail": f"Buy BTC @ {btc_ask:.2f}, Buy SOL @ {sol_btc_ask:.8f} BTC, Sell SOL @ {sol_bid:.2f}"},
            "route2": {"name": route2, "profit_pct": profit_2, "detail": f"Buy SOL @ {sol_ask:.2f}, Sell SOL @ {sol_btc_bid:.8f} BTC, Sell BTC @ {btc_bid:.2f}"},
        }, btc_bid, btc_ask, sol_bid, sol_ask, sol_btc_bid, sol_btc_ask
    except Exception as e:
        return None, None, None
