const http = require('http');
const fs = require('fs');
const path = require('path');

// Read credentials from environment (Replit Secrets)
const creds = {
  krakenKey: process.env.KRAKEN_API_KEY,
  krakenSecret: process.env.KRAKEN_SECRET,
  coinbaseKey: process.env.COINBASE_API_KEY,
  coinbaseSecret: process.env.COINBASE_SECRET,
};

console.log('✅ Auto Executor Ready');
console.log('Credentials loaded:', !!creds.krakenKey && !!creds.coinbaseKey);

// Webhook server listening for spike alerts
const server = http.createServer(async (req, res) => {
  if (req.method === 'POST' && req.url === '/execute') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', async () => {
      try {
        const alert = JSON.parse(body);
        const { coin, spread } = alert;
        
        console.log(`\n🔥 SPIKE ALERT: ${coin} at ${spread.toFixed(2)}% spread`);
        console.log('💰 Executing trade...');
        
        // Call your API's execute endpoint with credentials
        const payload = {
          krakenKey: creds.krakenKey,
          krakenSecret: creds.krakenSecret,
          coinbaseKey: creds.coinbaseKey,
          coinbaseSecret: creds.coinbaseSecret,
          coin: coin,
          tradeSizeUsd: 10,
          isDryRun: false
        };
        
        const apiReq = http.request({
          hostname: 'localhost',
          port: 5000,
          path: '/api/arb/ob-execute',
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Content-Length': Buffer.byteLength(JSON.stringify(payload))
          }
        }, (apiRes) => {
          let data = '';
          apiRes.on('data', chunk => data += chunk);
          apiRes.on('end', () => {
            try {
              const result = JSON.parse(data);
              const profit = result.realizedProfitUsd || 0;
              console.log(`✅ EXECUTED: +$${profit.toFixed(2)} profit`);
              
              // Log trade
              fs.appendFileSync('/tmp/trades_executed.jsonl', 
                JSON.stringify({
                  timestamp: new Date().toISOString(),
                  coin, spread, profit,
                  success: true
                }) + '\n'
              );
            } catch (e) {
              console.log('Trade result:', data);
            }
          });
        });
        
        apiReq.write(JSON.stringify(payload));
        apiReq.end();
        
        res.writeHead(200);
        res.end('OK');
      } catch (e) {
        console.error('Execution error:', e.message);
        res.writeHead(500);
        res.end('ERROR');
      }
    });
  } else {
    res.writeHead(404);
    res.end('Not found');
  }
});

server.listen(3001, () => {
  console.log('🚀 Webhook executor listening on http://localhost:3001/execute');
});
