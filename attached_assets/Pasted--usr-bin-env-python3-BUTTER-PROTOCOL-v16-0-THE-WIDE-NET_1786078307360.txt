def load_ledger():
    try:
        with open(LEDGER_FILE, "r") as f:
            data = json.load(f)
            return Decimal(data.get("profit", "0.00")), int(data.get("trades", 0))
    except:
        return Decimal("0.00"), 0

def save_ledger(profit, trades):
    with open(LEDGER_FILE, "w") as f:

# ---------- Kraken API Wrappers ----------
KRAKEN_BASE_URL = "https://api.kraken.com"
REQUEST_TIMEOUT = 10
session = requests.Session()

def D(v): return Decimal(str(v))

def kraken_private_request(api_key, api_secret, path, data):
    nonce = str(time.time_ns())
    payload = {"nonce": nonce, **data}
    postdata = urlencode(payload)
    encoded = (nonce + postdata).encode()
    message = path.encode() + hashlib.sha256(encoded).digest()
    decoded_secret = base64.b64decode(api_secret)
    api_sign = base64.b64encode(hmac.new(decoded_secret, message, hashlib.sha512).digest()).decode()
    resp = session.post(KRAKEN_BASE_URL + path, data=payload, headers={"API-Key": api_key, "API-Sign": api_sign}, timeout=REQUEST_TIMEOUT)
    resp.raise_for_status()
    result = resp.json()
    if result.get("error"):
        raise RuntimeError("Kraken: " + "; ".join(result["error"]))
    return result

def get_kraken_balance(api_key, api_secret):
    res = kraken_private_request(api_key, api_secret, "/0/private/Balance", {})
    return {k: D(v) for k, v in res["result"].items() if D(v) > 0}

def kraken_market_order(api_key, api_secret, pair, side, volume):
    return kraken_private_request(api_key, api_secret, "/0/private/AddOrder", {
        "pair": pair,
        "type": side.lower(),
        "ordertype": "market",
        "volume": format(volume, "f"),
    })

# ---------- ORDER BOOK FETCH & SIMULATION ----------
def fetch_order_book(pair, count=10):
    try:
        r = requests.get(f"https://api.kraken.com/0/public/Depth?pair={pair}&count={count}", timeout=3)
        data = r.json()
        if data.get("error"):
            return None
        res = list(data["result"].values())[0]
        asks = [[D(price), D(vol)] for price, vol in res["asks"]]
        bids = [[D(price), D(vol)] for price, vol in res["bids"]]
        return {"asks": asks, "bids": bids}
    except:
        return None

def simulate_triangular_cycle(asset_a, asset_b, start_usd, orderbooks, fees_pct):
    # ... (same simulation logic as v15, but I'll include it compactly)
    # Leg 1: Buy A with USD
    pair_a_usd = USD_PAIRS.get(asset_a)
    if not pair_a_usd or pair_a_usd not in orderbooks: return None
    ob_a_usd = orderbooks[pair_a_usd]
    if not ob_a_usd or not ob_a_usd["asks"]: return None
    remaining_usd = start_usd
    a_amt = D("0")
    avg_price_1 = D("0")
    best_price_1 = ob_a_usd["asks"][0][0]
    for price, vol in ob_a_usd["asks"]:
        cost = price * vol
        if remaining_usd <= cost:
            a_amt += remaining_usd / price
            avg_price_1 = (avg_price_1 * (a_amt - (remaining_usd / price)) + (remaining_usd / price) * price) / a_amt if a_amt > 0 else price
            remaining_usd = D("0")
            break
        else:
            a_amt += vol
            avg_price_1 = (avg_price_1 * (a_amt - vol) + vol * price) / a_amt
            remaining_usd -= cost
    if a_amt == 0 or remaining_usd > 0: return None

    # Leg 2: Sell A for B
    pair_a_b = CROSS_PAIRS.get((asset_a, asset_b))
    if not pair_a_b or pair_a_b not in orderbooks: return None
    ob_a_b = orderbooks[pair_a_b]
    if not ob_a_b or not ob_a_b["bids"]: return None
    b_amt = D("0")
    avg_price_2 = D("0")
    best_price_2 = ob_a_b["bids"][0][0]
    remaining_a = a_amt
    for price, vol in ob_a_b["bids"]:
        if remaining_a <= vol:
            b_amt += remaining_a * price
            avg_price_2 = (avg_price_2 * (b_amt - (remaining_a * price)) + (remaining_a * price)) / b_amt if b_amt > 0 else price
            remaining_a = D("0")
            break
        else:
            b_amt += vol * price
            avg_price_2 = (avg_price_2 * (b_amt - (vol * price)) + (vol * price)) / b_amt
            remaining_a -= vol
    if b_amt == 0 or remaining_a > 0: return None

    # Leg 3: Sell B for USD
    pair_b_usd = USD_PAIRS.get(asset_b)
    if not pair_b_usd or pair_b_usd not in orderbooks: return None
    ob_b_usd = orderbooks[pair_b_usd]
    if not ob_b_usd or not ob_b_usd["bids"]: return None
    usd_final = D("0")
    avg_price_3 = D("0")
    best_price_3 = ob_b_usd["bids"][0][0]
    remaining_b = b_amt
    for price, vol in ob_b_usd["bids"]:
        if remaining_b <= vol:
            usd_final += remaining_b * price
            avg_price_3 = (avg_price_3 * (usd_final - (remaining_b * price)) + (remaining_b * price)) / usd_final if usd_final > 0 else price
            remaining_b = D("0")
            break
        else:
            usd_final += vol * price
            avg_price_3 = (avg_price_3 * (usd_final - (vol * price)) + (vol * price)) / usd_final
            remaining_b -= vol
    if usd_final == 0 or remaining_b > 0: return None

    profit = usd_final - start_usd
    profit = profit * (D("1") - fees_pct / D("100"))

    slippage_1 = abs(avg_price_1 - best_price_1) / best_price_1 * 100 if best_price_1 > 0 else 0
    slippage_2 = abs(avg_price_2 - best_price_2) / best_price_2 * 100 if best_price_2 > 0 else 0
    slippage_3 = abs(avg_price_3 - best_price_3) / best_price_3 * 100 if best_price_3 > 0 else 0
    total_slippage = slippage_1 + slippage_2 + slippage_3

    return profit, avg_price_1, avg_price_2, avg_price_3, a_amt, total_slippage

            # 2. Simulate all cycles (10 assets = 90 routes)
            cycles = []
            for asset_a, asset_b in permutations(ASSETS, 2):
                pair_a_usd = USD_PAIRS.get(asset_a)

                if pair_a_usd not in orderbooks or pair_b_usd not in orderbooks or pair_a_b not in orderbooks:

                if not result:
                    continue
                profit, avg1, avg2, avg3, vol_a, slippage = result

                    "asset_a": asset_a,
                    "asset_b": asset_b,
                    "profit": profit,
                    "slippage": slippage,
                    "vol": vol_a,
                    "status": status,

                        # --- PRE-FLIGHT CHECK ---
                        add_log(f"🔍 Pre-flight check for {best_ready['route']}...")
                        fresh_books = {}
                        pairs_needed = [USD_PAIRS[best_ready["asset_a"]], USD_PAIRS[best_ready["asset_b"]], CROSS_PAIRS[(best_ready["asset_a"], best_ready["asset_b"])]]
                        for p in pairs_needed:

                            if ob:
                                fresh_books[p] = ob
                        if all(p in fresh_books for p in pairs_needed):
                            fresh_result = simulate_triangular_cycle(
                                best_ready["asset_a"], best_ready["asset_b"],
                                trade_size_usd, fresh_books, tri_fees
                            )
                            if fresh_result:

                                            st.session_state.trade_count += 1
                                            st.session_state.last_trade_time = time.time()
                                            save_ledger(st.session_state.total_profit_usd, st.session_state.trade_count)
