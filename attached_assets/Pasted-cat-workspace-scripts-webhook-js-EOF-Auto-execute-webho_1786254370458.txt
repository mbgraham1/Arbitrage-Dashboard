// Auto-execute webhook for CAT_ARB
const http = require('http');
const fs = require('fs');

const PORT = 3001;

const server = http.createServer(async (req, res) => {
  if (req.method === 'POST' && req.url === '/spike') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', async () => {
      try {
        const { coin, spread } = JSON.parse(body);
        console.log(`\n🔥 Spike: ${coin} @ ${spread.toFixed(2)}%`);
        
        const payload = JSON.stringify({
          krakenKey: process.env.KRAKEN_API_KEY,
          krakenSecret: process.env.KRAKEN_SECRET,
          coinbaseKey: process.env.COINBASE_API_KEY,
          coinbaseSecret: process.env.COINBASE_SECRET,
          coin: coin,
          tradeSizeUsd: 10,
          isDryRun: false
        });
        
        const options = {
          hostname: 'localhost',
          port: 5000,
          path: '/api/arb/ob-execute',
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Content-Length': Buffer.byteLength(payload)
          }
        };
        
        const apiReq = http.request(options, (apiRes) => {
          let data = '';
          apiRes.on('data', chunk => data += chunk);
          apiRes.on('end', () => {
            const result = JSON.parse(data);
            if (result.success) {
              console.log(`✅ ${coin}: +$${(result.realizedProfitUsd || 0).toFixed(2)}`);
            } else {
              console.log(`⚠️ ${coin}: Rejected`);
            }
          });
        });
        
        apiReq.write(payload);
        apiReq.end();
        
        res.writeHead(200);
        res.end('OK');
      } catch (e) {
        res.writeHead(400);
        res.end('Error');
      }
    });
  } else {
    res.writeHead(404);
    res.end();
  }
});

server.listen(PORT, () => {
  console.log(`✅ Webhook on port ${PORT}`);
  console.log(`✅ Forwarding to your API (all security checks intact)`);
});
